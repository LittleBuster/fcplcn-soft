cmake_minimum_required(VERSION 3.22)
project(fcplcn_board_profile_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON source directory")
if(NOT EXISTS "${FCPLCN_CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Configure firmware/base or set FCPLCN_CJSON_DIR to the managed cJSON directory")
endif()
add_library(fcplcn_profile_cjson STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_profile_cjson PUBLIC "${FCPLCN_CJSON_DIR}")
enable_testing()
foreach(variant current eeprom)
    add_executable(fcplcn_profile_${variant}_tests profile_test.cpp
        "${FCPLCN_BASE_DIR}/main/hardware/board_profile.cpp")
    target_include_directories(fcplcn_profile_${variant}_tests PRIVATE "${FCPLCN_BASE_DIR}/main")
    target_link_libraries(fcplcn_profile_${variant}_tests PRIVATE fcplcn_profile_cjson)
    target_compile_options(fcplcn_profile_${variant}_tests PRIVATE -Wall -Wextra -Werror)
    if(variant STREQUAL "eeprom")
        target_compile_definitions(fcplcn_profile_${variant}_tests PRIVATE
            FCPLCN_BOOT_BOARD_ID="fcplc-080806-3v1")
    endif()
    add_test(NAME "fcplcn_profile_${variant}_boot" COMMAND fcplcn_profile_${variant}_tests)
    set_tests_properties("fcplcn_profile_${variant}_boot" PROPERTIES TIMEOUT 10)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME fcplcn_profile_packaging
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/package_test.py")
set_tests_properties(fcplcn_profile_packaging PROPERTIES TIMEOUT 10)
