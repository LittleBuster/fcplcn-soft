cmake_minimum_required(VERSION 3.22)
project(fcplcn_hardware_io_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON source directory")
if(NOT EXISTS "${FCPLCN_CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Configure firmware/base or set FCPLCN_CJSON_DIR to the managed cJSON directory")
endif()
find_package(Threads REQUIRED)
add_library(fcplcn_io_cjson STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_io_cjson PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_io_tests io_test.cpp stubs/driver.cpp
    "${FCPLCN_BASE_DIR}/main/hardware/hardware_io.cpp" "${FCPLCN_BASE_DIR}/main/hardware/board_profile.cpp"
    "${FCPLCN_BASE_DIR}/main/hardware/fram.cpp")
target_include_directories(fcplcn_io_tests PRIVATE stubs "${FCPLCN_BASE_DIR}/main")
target_compile_definitions(fcplcn_io_tests PRIVATE FCPLCN_HARDWARE_IO_TESTING=1)
target_compile_options(fcplcn_io_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(fcplcn_io_tests PRIVATE fcplcn_io_cjson Threads::Threads)
option(FCPLCN_TEST_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_io_cjson fcplcn_io_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_io_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case fake_map safe_init relay_rmw raw_inputs failures missing_recovery scans concurrent
             emi_reset emi_layout emi_fields emi_readback emi_repair_faults off_revocation
             failed_on recovery_visibility controller_policy json_stale
             fram_transfer fram_bounds fram_failures fram_concurrent)
    add_test(NAME "fcplcn_io_${case}" COMMAND fcplcn_io_tests "${case}")
    set_tests_properties("fcplcn_io_${case}" PROPERTIES TIMEOUT 20)
endforeach()
