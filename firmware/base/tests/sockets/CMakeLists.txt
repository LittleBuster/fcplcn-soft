cmake_minimum_required(VERSION 3.22)
project(fcplcn_socket_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON directory")
find_package(Threads REQUIRED)
include("${FCPLCN_BASE_DIR}/main/controllers/sources.cmake")
add_library(fcplcn_socket_json STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_socket_json PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_socket_tests socket_test.cpp stubs/platform.cpp ../stubs/settings_store.cpp ../stubs/meteo_sensors.cpp ../stubs/ibutton.cpp
    ${FCPLCN_CONTROLLER_SOURCES} "${FCPLCN_BASE_DIR}/main/config/state_journal.cpp" "${FCPLCN_BASE_DIR}/main/hardware/board_profile.cpp"
    "${FCPLCN_BASE_DIR}/main/network/socket_mqtt.cpp" "${FCPLCN_BASE_DIR}/main/config/model.cpp")
target_include_directories(fcplcn_socket_tests PRIVATE stubs ../hardware_io/stubs ../stubs "${FCPLCN_BASE_DIR}/main")
target_compile_definitions(fcplcn_socket_tests PRIVATE FCPLCN_SOCKET_CONTROLLER_TESTING=1)
target_compile_options(fcplcn_socket_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(fcplcn_socket_tests PRIVATE fcplcn_socket_json Threads::Threads)
option(FCPLCN_TEST_SANITIZERS "Enable ASan/UBSan" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_socket_json fcplcn_socket_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_socket_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case defaults persistence edges exclusive_inputs validation reconfigure disable storage_failure hardware_failure recovery concurrent corrupt_storage mqtt_set mqtt_replay startup_low input_failure mqtt_snapshot lights_basic resource_exclusivity concurrent_claims lights_mqtt meteo_basic meteo_resources meteo_readings meteo_scan thermostat_heating thermostat_cooling_shared thermostat_faults thermostat_source_binding thermostat_validation thermostat_safe_config thermostat_hardware_recovery thermostat_expiry thermostat_failed_update_expiry thermostat_item_switch thermostat_item_buttons thermostat_item_input_fault thermostat_item_claims_config thermostat_item_storage_inhibit thermostat_item_expiry thermostat_item_mqtt thermostat_recovery_policy thermostat_dwell thermostat_dwell_defaults thermostat_cached_sample button_confirmation fram_states fram_safe_boot fram_config_transaction fram_thermostat_flags fram_diagnostics security_module_enable security_optional_hardware security_zones security_keys security_delays security_persistence security_faults security_enrollment security_claims security_deadlines security_entry_reboot security_mqtt security_config_held_key security_hub_touch security_hub_freshness security_hub_fence security_hub_busy security_hub_failure_event security_hub_contact_before_lease)
    add_test(NAME fcplcn_sockets_${case} COMMAND fcplcn_socket_tests ${case})
    set_tests_properties(fcplcn_sockets_${case} PROPERTIES TIMEOUT 30)
endforeach()
foreach(case legacy_buttons legacy_button_recovery legacy_thermostat_button legacy_security_zones legacy_security_recovery)
    add_test(NAME fcplcn_sockets_${case} COMMAND fcplcn_socket_tests ${case})
    set_tests_properties(fcplcn_sockets_${case} PROPERTIES TIMEOUT 30)
endforeach()
