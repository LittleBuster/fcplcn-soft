cmake_minimum_required(VERSION 3.22)
project(fcplcn_ota_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON")
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(fcplcn_ota_json STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_ota_json PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_ota_tests ota_test.cpp stubs/platform.cpp "${FCPLCN_BASE_DIR}/main/system/firmware_update.cpp")
target_include_directories(fcplcn_ota_tests PRIVATE stubs ../backup/stubs "${FCPLCN_BASE_DIR}/main")
file(READ "${FCPLCN_BASE_DIR}/../../config/product.json" FCPLCN_IDENTITY_JSON)
string(JSON FCPLCN_PRODUCT_NAME GET "${FCPLCN_IDENTITY_JSON}" name)
string(JSON FCPLCN_PRODUCT_FULL_NAME GET "${FCPLCN_IDENTITY_JSON}" fullName)
configure_file("${FCPLCN_BASE_DIR}/main/system/product_identity.h.in" generated/product_identity.h @ONLY)
target_include_directories(fcplcn_ota_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(fcplcn_ota_tests PRIVATE FCPLCN_FIRMWARE_UPDATE_TESTING=1)
target_compile_options(fcplcn_ota_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(fcplcn_ota_tests PRIVATE fcplcn_ota_json OpenSSL::Crypto Threads::Threads)
option(FCPLCN_TEST_SANITIZERS "Enable ASan/UBSan" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_ota_json fcplcn_ota_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_ota_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case happy begin identity chunks ordering hash flash expiry deadline reboot incomplete)
    add_test(NAME fcplcn_ota_${case} COMMAND fcplcn_ota_tests ${case})
    set_tests_properties(fcplcn_ota_${case} PROPERTIES TIMEOUT 15)
endforeach()
