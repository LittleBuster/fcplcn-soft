cmake_minimum_required(VERSION 3.22)
project(fcplcn_mqtt_telemetry_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON directory")
add_library(fcplcn_telemetry_json STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_telemetry_json PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_telemetry_tests telemetry_test.cpp "${FCPLCN_BASE_DIR}/main/network/mqtt_telemetry.cpp")
target_include_directories(fcplcn_telemetry_tests PRIVATE "${FCPLCN_BASE_DIR}/main")
target_link_libraries(fcplcn_telemetry_tests PRIVATE fcplcn_telemetry_json)
target_compile_options(fcplcn_telemetry_tests PRIVATE -Wall -Wextra -Werror)
option(FCPLCN_TEST_SANITIZERS "Enable ASan/UBSan" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_telemetry_json fcplcn_telemetry_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_telemetry_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case ages fresh_sample faults retry_reconnect refresh bounds dwell large_discovery security tanks irrigation)
    add_test(NAME fcplcn_telemetry_${case} COMMAND fcplcn_telemetry_tests ${case})
    set_tests_properties(fcplcn_telemetry_${case} PROPERTIES TIMEOUT 15)
endforeach()
