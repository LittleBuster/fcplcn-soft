cmake_minimum_required(VERSION 3.22)
project(fcplcn_backup_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON directory")
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
include("${FCPLCN_BASE_DIR}/main/controllers/sources.cmake")
add_library(fcplcn_backup_json STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_backup_json PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_backup_tests backup_test.cpp stubs/platform.cpp stubs/crypto.cpp ../stubs/meteo_sensors.cpp ../stubs/ibutton.cpp
    "${FCPLCN_BASE_DIR}/main/config/backup.cpp" "${FCPLCN_BASE_DIR}/main/config/settings_store.cpp"
    "${FCPLCN_BASE_DIR}/main/config/config.cpp" "${FCPLCN_BASE_DIR}/main/config/model.cpp"
    "${FCPLCN_BASE_DIR}/main/security/auth.cpp" ${FCPLCN_CONTROLLER_SOURCES} "${FCPLCN_BASE_DIR}/main/config/state_journal.cpp"
    "${FCPLCN_BASE_DIR}/main/hardware/board_profile.cpp")
target_include_directories(fcplcn_backup_tests PRIVATE stubs ../auth/stubs ../hardware_io/stubs ../stubs "${FCPLCN_BASE_DIR}/main")
file(READ "${FCPLCN_BASE_DIR}/../../config/product.json" FCPLCN_IDENTITY_JSON)
string(JSON FCPLCN_PRODUCT_NAME GET "${FCPLCN_IDENTITY_JSON}" name)
string(JSON FCPLCN_PRODUCT_FULL_NAME GET "${FCPLCN_IDENTITY_JSON}" fullName)
configure_file("${FCPLCN_BASE_DIR}/main/system/product_identity.h.in" generated/product_identity.h @ONLY)
target_include_directories(fcplcn_backup_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(fcplcn_backup_tests PRIVATE FCPLCN_SETTINGS_TESTING=1 FCPLCN_SOCKET_CONTROLLER_TESTING=1)
target_compile_options(fcplcn_backup_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(fcplcn_backup_tests PRIVATE fcplcn_backup_json OpenSSL::Crypto Threads::Threads)
option(FCPLCN_TEST_SANITIZERS "Enable ASan/UBSan" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_backup_json fcplcn_backup_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_backup_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case defaults clone rejection failed_commit cas pending migration migration_failure corrupt_store maximum gate allocation_failure legacy_profile legacy_backup lights_clone resource_rejection legacy_v1 legacy_conflict meteo_clone meteo_rejection legacy_v2_storage legacy_v2_backup maximum_meteo thermostat_clone thermostat_rejection thermostat_missing_source legacy_v3_storage legacy_v3_backup maximum_thermostats thermostat_modes_clone legacy_v4_storage legacy_v4_backup legacy_v5_storage legacy_v5_backup legacy_v6_storage legacy_v6_backup thermostat_policy_import legacy_v7_storage legacy_v7_backup legacy_v8_storage legacy_v8_missing_fram legacy_v8_migration_failure security_clone security_rejection security_atomic_failure security_entry_backup fram_export_restore fram_restore_failure fram_export_fault_recovery)
    add_test(NAME fcplcn_backup_${case} COMMAND fcplcn_backup_tests ${case})
    set_tests_properties(fcplcn_backup_${case} PROPERTIES TIMEOUT 45)
endforeach()
