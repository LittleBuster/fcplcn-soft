cmake_minimum_required(VERSION 3.22)
project(fcplcn_runtime_policy_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(fcplcn_runtime_policy_tests runtime_policy_test.cpp)
target_include_directories(fcplcn_runtime_policy_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
target_compile_options(fcplcn_runtime_policy_tests PRIVATE -Wall -Wextra -Werror)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(fcplcn_runtime_policy_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fcplcn_runtime_policy_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME fcplcn_runtime_watchdog_policy COMMAND fcplcn_runtime_policy_tests)
set_tests_properties(fcplcn_runtime_watchdog_policy PROPERTIES TIMEOUT 15)
