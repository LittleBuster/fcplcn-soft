cmake_minimum_required(VERSION 3.22)
project(fcplcn_management_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FCPLCN_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(FCPLCN_CJSON_DIR "${FCPLCN_BASE_DIR}/managed_components/espressif__cjson/cJSON" CACHE PATH "Managed cJSON directory")
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
include("${FCPLCN_BASE_DIR}/main/controllers/sources.cmake")
add_library(fcplcn_management_json STATIC "${FCPLCN_CJSON_DIR}/cJSON.c")
target_include_directories(fcplcn_management_json PUBLIC "${FCPLCN_CJSON_DIR}")
add_executable(fcplcn_management_tests management_test.cpp stubs/services.cpp
    ../backup/stubs/platform.cpp ../backup/stubs/crypto.cpp ../stubs/meteo_sensors.cpp ../stubs/ibutton.cpp
    "${FCPLCN_BASE_DIR}/main/config/backup.cpp" "${FCPLCN_BASE_DIR}/main/config/settings_store.cpp"
    "${FCPLCN_BASE_DIR}/main/config/config.cpp" "${FCPLCN_BASE_DIR}/main/config/model.cpp"
    "${FCPLCN_BASE_DIR}/main/security/auth.cpp" ${FCPLCN_CONTROLLER_SOURCES}
    "${FCPLCN_BASE_DIR}/main/config/state_journal.cpp" "${FCPLCN_BASE_DIR}/main/hardware/board_profile.cpp"
    "${FCPLCN_BASE_DIR}/main/network/management_mqtt.cpp" "${FCPLCN_BASE_DIR}/main/network/management_routes.cpp")
target_include_directories(fcplcn_management_tests PRIVATE stubs ../backup/stubs ../auth/stubs ../hardware_io/stubs ../stubs "${FCPLCN_BASE_DIR}/main")
file(READ "${FCPLCN_BASE_DIR}/../../config/product.json" FCPLCN_IDENTITY_JSON)
string(JSON FCPLCN_PRODUCT_NAME GET "${FCPLCN_IDENTITY_JSON}" name)
string(JSON FCPLCN_PRODUCT_FULL_NAME GET "${FCPLCN_IDENTITY_JSON}" fullName)
configure_file("${FCPLCN_BASE_DIR}/main/system/product_identity.h.in" generated/product_identity.h @ONLY)
target_include_directories(fcplcn_management_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(fcplcn_management_tests PRIVATE FCPLCN_SETTINGS_TESTING=1 FCPLCN_SOCKET_CONTROLLER_TESTING=1)
target_compile_options(fcplcn_management_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(fcplcn_management_tests PRIVATE fcplcn_management_json OpenSSL::Crypto Threads::Threads)
option(FCPLCN_TEST_SANITIZERS "Enable ASan/UBSan" ON)
if(FCPLCN_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target fcplcn_management_json fcplcn_management_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(fcplcn_management_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case reads validation replay config controllers ownership scans user time backup restore fragments large generation)
    add_test(NAME fcplcn_management_${case} COMMAND fcplcn_management_tests ${case})
    set_tests_properties(fcplcn_management_${case} PROPERTIES TIMEOUT 45)
endforeach()
